state_wis <- read.csv("C:/Projects/ForecastEvaluation/Data/state-scoring/state_wis.csv")
standard_wis <- read.csv("C:/Projects/ForecastEvaluation/Data/standard-scoring/standard_wis.csv")
state_wis = state_wis |>
    rename(state_wis = wis)
complete_wis <- state_wis %>%
    left_join(standard_wis, by = intersect(colnames(state_wis),
        colnames(standard_wis)))

complete_wis$relative_wis <- complete_wis$state_wis/complete_wis$wis
complete_wis = complete_wis |>
    select("location", "target_end_date", "horizon", "reference_date",
        "model_id", "target", "wis", "state_wis", "relative_wis")

complete_wis <- complete_wis %>%
    filter(wis > 0)

temp2a <- model_effects %>%
    arrange(-effect) %>%
    mutate(model_id = factor(model_id, levels = model_id))

fig2a <- ggplot(temp2a, aes(x = effect, y = model_id)) + geom_point(size = 4,
    color = "steelblue") + labs(x = "Effect (log WIS)", y = "Model") +
    theme_bw() + theme(text = element_text(size = 18), axis.text.x = element_text(size = 16),
    axis.text.y = element_text(size = 16), axis.title = element_text(size = 18),
    legend.text = element_text(size = 16), legend.title = element_text(size = 17),
    strip.text = element_text(size = 17)) + theme(legend.position = "top") +
    geom_vline(aes(xintercept = 0), col = "black", linetype = "dashed",
        linewidth = 1.2)
temp2b <- location_effects %>%
    arrange(-effect) %>%
    mutate(location = factor(location, levels = location))

fig2b <- ggplot(temp2b, aes(x = effect, y = location)) + geom_point(size = 4,
    color = "coral") + labs(x = "Effect (log WIS)", y = "Location") +
    theme_bw() + theme(text = element_text(size = 18), axis.text.x = element_text(size = 16),
    axis.text.y = element_text(size = 16), axis.title = element_text(size = 18),
    legend.text = element_text(size = 16), legend.title = element_text(size = 17),
    strip.text = element_text(size = 17)) + theme(legend.position = "top") +
    geom_vline(aes(xintercept = 0), col = "black", linetype = "dashed",
        linewidth = 1.2)
fig2a + fig2b + plot_layout(ncol = 2, widths = c(0.4, 0.6))
